-- ---------------------------------------------------------------------
-- 1. USERS  (base table — shared by Patient, Doctor, Admin)
-- ---------------------------------------------------------------------
CREATE TABLE users (
    user_id         INT AUTO_INCREMENT PRIMARY KEY,
    name            VARCHAR(100)        NOT NULL,
    email           VARCHAR(150)        NOT NULL UNIQUE,
    password_hash   VARCHAR(255)        NOT NULL,
    phone           VARCHAR(20),
    role            ENUM('patient', 'doctor', 'admin') NOT NULL,
    created_at      DATETIME            NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME            NOT NULL DEFAULT CURRENT_TIMESTAMP
                                          ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 2. PATIENTS  (extends users — role-specific fields)
-- ---------------------------------------------------------------------
CREATE TABLE patients (
    patient_id      INT PRIMARY KEY,               -- same value as users.user_id
    blood_group     VARCHAR(5),
    date_of_birth   DATE,
    address         VARCHAR(255),

    CONSTRAINT fk_patients_user
        FOREIGN KEY (patient_id) REFERENCES users(user_id)
        ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 3. DOCTORS  (extends users — role-specific fields)
-- ---------------------------------------------------------------------
CREATE TABLE doctors (
    doctor_id           INT PRIMARY KEY,           -- same value as users.user_id
    specialization      VARCHAR(100)    NOT NULL,
    hospital            VARCHAR(150),
    consultation_fee    DECIMAL(10,2)   NOT NULL DEFAULT 0.00,
    rating              DECIMAL(2,1)    DEFAULT 0.0,
    is_verified         BOOLEAN         NOT NULL DEFAULT FALSE,
    verified_by         INT             NULL,       -- admin who verified them
    verified_at         DATETIME        NULL,
    bio                 TEXT,

    CONSTRAINT fk_doctors_user
        FOREIGN KEY (doctor_id) REFERENCES users(user_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_doctors_verified_by
        FOREIGN KEY (verified_by) REFERENCES users(user_id)
        ON DELETE SET NULL
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 4. ADMINS  (extends users — role-specific fields)
-- ---------------------------------------------------------------------
CREATE TABLE admins (
    admin_id        INT PRIMARY KEY,               -- same value as users.user_id
    admin_role      VARCHAR(50) DEFAULT 'general',  -- e.g. 'super', 'support'

    CONSTRAINT fk_admins_user
        FOREIGN KEY (admin_id) REFERENCES users(user_id)
        ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 5. TIME SLOTS  (a doctor's bookable availability)
-- ---------------------------------------------------------------------
CREATE TABLE time_slots (
    slot_id         INT AUTO_INCREMENT PRIMARY KEY,
    doctor_id       INT             NOT NULL,
    slot_date       DATE            NOT NULL,
    start_time      TIME            NOT NULL,
    end_time        TIME            NOT NULL,
    is_booked       BOOLEAN         NOT NULL DEFAULT FALSE,

    CONSTRAINT fk_slots_doctor
        FOREIGN KEY (doctor_id) REFERENCES doctors(doctor_id)
        ON DELETE CASCADE,
    UNIQUE KEY uq_doctor_slot (doctor_id, slot_date, start_time)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 6. APPOINTMENTS
-- ---------------------------------------------------------------------
CREATE TABLE appointments (
    appointment_id  INT AUTO_INCREMENT PRIMARY KEY,
    patient_id      INT             NOT NULL,
    doctor_id       INT             NOT NULL,
    slot_id         INT             NOT NULL UNIQUE,   -- one slot -> one appointment
    status          ENUM('pending', 'confirmed', 'completed', 'cancelled')
                                    NOT NULL DEFAULT 'pending',
    fee             DECIMAL(10,2)   NOT NULL,
    created_at      DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP
                                    ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT fk_appt_patient
        FOREIGN KEY (patient_id) REFERENCES patients(patient_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_appt_doctor
        FOREIGN KEY (doctor_id) REFERENCES doctors(doctor_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_appt_slot
        FOREIGN KEY (slot_id) REFERENCES time_slots(slot_id)
        ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 7. PAYMENTS  (0..1 per appointment)
-- ---------------------------------------------------------------------
CREATE TABLE payments (
    payment_id      INT AUTO_INCREMENT PRIMARY KEY,
    appointment_id  INT             NOT NULL UNIQUE,
    amount          DECIMAL(10,2)   NOT NULL,
    method          ENUM('bkash', 'card', 'nagad') NOT NULL,
    status          ENUM('pending', 'completed', 'failed', 'refunded')
                                    NOT NULL DEFAULT 'pending',
    transaction_id  VARCHAR(100),
    paid_at         DATETIME        NULL,
    created_at      DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_payment_appt
        FOREIGN KEY (appointment_id) REFERENCES appointments(appointment_id)
        ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 8. PRESCRIPTIONS  (0..1 per appointment)
-- ---------------------------------------------------------------------
CREATE TABLE prescriptions (
    prescription_id INT AUTO_INCREMENT PRIMARY KEY,
    appointment_id  INT             NOT NULL UNIQUE,
    doctor_id       INT             NOT NULL,
    patient_id      INT             NOT NULL,
    medicines       TEXT            NOT NULL,       -- e.g. "Napa 500, Seclo 20"
    notes           TEXT,
    issued_date     DATE            NOT NULL,

    CONSTRAINT fk_presc_appt
        FOREIGN KEY (appointment_id) REFERENCES appointments(appointment_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_presc_doctor
        FOREIGN KEY (doctor_id) REFERENCES doctors(doctor_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_presc_patient
        FOREIGN KEY (patient_id) REFERENCES patients(patient_id)
        ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 9. MEDICAL RECORDS  (a patient's visit history)
-- ---------------------------------------------------------------------
CREATE TABLE medical_records (
    record_id       INT AUTO_INCREMENT PRIMARY KEY,
    patient_id      INT             NOT NULL,
    doctor_id       INT             NOT NULL,
    appointment_id  INT             NULL,
    diagnosis       VARCHAR(255)    NOT NULL,
    notes           TEXT,
    visit_date      DATE            NOT NULL,

    CONSTRAINT fk_record_patient
        FOREIGN KEY (patient_id) REFERENCES patients(patient_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_record_doctor
        FOREIGN KEY (doctor_id) REFERENCES doctors(doctor_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_record_appt
        FOREIGN KEY (appointment_id) REFERENCES appointments(appointment_id)
        ON DELETE SET NULL
) ENGINE=InnoDB;

-- =====================================================================
-- AI AGENTS
-- =====================================================================

-- ---------------------------------------------------------------------
-- 10. AI_AGENTS  (registry of all AI agents — implemented or planned)
-- ---------------------------------------------------------------------
CREATE TABLE ai_agents (
    agent_id        INT AUTO_INCREMENT PRIMARY KEY,
    name            VARCHAR(100)    NOT NULL,
    type            ENUM('medicine_assistant', 'doctor_matching',
                         'appointment_reminder', 'triage')
                                    NOT NULL UNIQUE,
    model_name      VARCHAR(100)    NOT NULL DEFAULT 'gemini-pro',
    status          ENUM('active', 'planned') NOT NULL DEFAULT 'planned'
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 11. MEDICINE_QUERIES  (log of Medicine Assistant lookups)
-- ---------------------------------------------------------------------
CREATE TABLE medicine_queries (
    query_id        INT AUTO_INCREMENT PRIMARY KEY,
    patient_id      INT             NOT NULL,
    agent_id        INT             NOT NULL,
    medicine_name   VARCHAR(150)    NOT NULL,
    response        TEXT,
    queried_at      DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_medquery_patient
        FOREIGN KEY (patient_id) REFERENCES patients(patient_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_medquery_agent
        FOREIGN KEY (agent_id) REFERENCES ai_agents(agent_id)
        ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 12. DOCTOR_MATCH_SUGGESTIONS  (Doctor Matching Agent — planned)
-- ---------------------------------------------------------------------
CREATE TABLE doctor_match_suggestions (
    suggestion_id       INT AUTO_INCREMENT PRIMARY KEY,
    patient_id          INT         NOT NULL,
    agent_id            INT         NOT NULL,
    symptoms            TEXT        NOT NULL,
    suggested_doctor_id INT         NULL,
    created_at          DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_match_patient
        FOREIGN KEY (patient_id) REFERENCES patients(patient_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_match_agent
        FOREIGN KEY (agent_id) REFERENCES ai_agents(agent_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_match_doctor
        FOREIGN KEY (suggested_doctor_id) REFERENCES doctors(doctor_id)
        ON DELETE SET NULL
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 13. APPOINTMENT_REMINDERS  (Appointment Reminder Agent — planned)
-- ---------------------------------------------------------------------
CREATE TABLE appointment_reminders (
    reminder_id     INT AUTO_INCREMENT PRIMARY KEY,
    appointment_id  INT             NOT NULL,
    agent_id        INT             NOT NULL,
    reminder_type   ENUM('before', 'followup') NOT NULL,
    status          ENUM('sent', 'failed') NOT NULL DEFAULT 'sent',
    sent_at         DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_reminder_appt
        FOREIGN KEY (appointment_id) REFERENCES appointments(appointment_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_reminder_agent
        FOREIGN KEY (agent_id) REFERENCES ai_agents(agent_id)
        ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 14. TRIAGE_SESSIONS  (Triage Assistant — planned)
-- ---------------------------------------------------------------------
CREATE TABLE triage_sessions (
    session_id              INT AUTO_INCREMENT PRIMARY KEY,
    patient_id               INT        NOT NULL,
    agent_id                  INT        NOT NULL,
    answers                    JSON,
    recommended_specialty      VARCHAR(100),
    created_at                 DATETIME   NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_triage_patient
        FOREIGN KEY (patient_id) REFERENCES patients(patient_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_triage_agent
        FOREIGN KEY (agent_id) REFERENCES ai_agents(agent_id)
        ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 15. ACTIVITY_LOGS  (for the admin "Manage users & reports" screen)
-- ---------------------------------------------------------------------
CREATE TABLE activity_logs (
    log_id          INT AUTO_INCREMENT PRIMARY KEY,
    admin_id        INT             NOT NULL,
    action          VARCHAR(150)    NOT NULL,   -- e.g. 'verified_doctor'
    target_type     VARCHAR(50),                -- e.g. 'doctor', 'user'
    target_id       INT,
    created_at      DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_log_admin
        FOREIGN KEY (admin_id) REFERENCES admins(admin_id)
        ON DELETE CASCADE
) ENGINE=InnoDB;
